    78→// ── Order Status Stepper ─────────────────────────────────────
    79→function OrderStepper({ currentStatus }: { currentStatus: ClientOrderStatus }) {
    80→  const currentStep = STATUS_CONFIG[currentStatus].step;
    81→  const steps = [
    82→    { num: 1, label: 'Commandé', icon: <ShoppingBag className="w-3 h-3" /> },
    83→    { num: 2, label: 'Payé', icon: <CreditCard className="w-3 h-3" /> },
    84→    { num: 3, label: 'Expédié', icon: <Truck className="w-3 h-3" /> },
    85→    { num: 4, label: 'Livré', icon: <Package className="w-3 h-3" /> },
    86→  ];
    87→
    88→  return (
    89→    <div className="flex items-center justify-between">
    90→      {steps.map((step, i) => {
    91→        const isDone = step.num <= currentStep;
    92→        const isCurrent = step.num === currentStep;
    93→        return (
    94→          <div key={step.num} className="flex items-center flex-1 last:flex-none">
    95→            <div className="flex flex-col items-center gap-1">
    96→              <div
    97→                className={`w-7 h-7 rounded-full flex items-center justify-center transition-all duration-500 ${
    98→                  isDone
    99→                    ? isCurrent
   100→                      ? 'bg-orange-500 text-white ring-4 ring-orange-100'
   101→                      : 'bg-emerald-500 text-white'
   102→                    : 'bg-gray-100 text-gray-300'
   103→                }`}
   104→              >
   105→                {isDone && !isCurrent ? (
   106→                  <CheckCircle2 className="w-3.5 h-3.5" />
   107→                ) : (
   108→                  step.icon
   109→                )}
   110→              </div>
   111→              <span
   112→                className={`text-[9px] font-medium transition-colors ${
   113→                  isDone ? 'text-gray-700' : 'text-gray-300'
   114→                }`}
   115→              >
   116→                {step.label}
   117→              </span>
   118→            </div>
   119→            {i < steps.length - 1 && (
   120→              <div
   121→                className={`flex-1 h-0.5 mx-1 mb-4 transition-colors duration-500 ${
   122→                  step.num < currentStep ? 'bg-emerald-300' : 'bg-gray-100'
   123→                }`}
   124→              />
   125→            )}
   126→          </div>
   127→        );
   128→      })}
   129→    </div>
   130→  );
   131→}

   534→            Oui, j&apos;ai bien reçu ma commande
   535→          </Button>
   536→          <button
   537→            onClick={onCancel}
   538→            className="w-full py-3 text-sm text-gray-400 hover:text-gray-600 transition-colors"
   539→          >
   540→            Pas encore, annuler
   541→          </button>
   542→        </div>
   543→      </div>
   544→    </div>

   548→// ── Main Component ───────────────────────────────────────────
   549→type FilterKey = 'all' | 'active' | 'delivered';
   550→
   551→export function ClientOrders() {
   552→  const {
   553→    clientOrders,
   554→    activeClientOrderFilter,
   555→    setClientOrderFilter,
   556→    confirmReceipt,
   557→    confirmingReceiptId,
   558→    setConfirmingReceiptId,
   559→  } = useAppStore();
   560→
   561→  const [expandedOrder, setExpandedOrder] = useState<string | null>(null);
   562→  const [showSuccessToast, setShowSuccessToast] = useState(false);
   563→
   564→  // Filter orders
   565→  const filteredOrders = clientOrders.filter((order) => {
   566→    if (activeClientOrderFilter === 'active') {
   567→      return order.status !== 'delivered';
   568→    }
   569→    if (activeClientOrderFilter === 'delivered') {
   570→      return order.status === 'delivered';
   571→    }
   572→    return true;
   573→  });
   574→
   575→  // Counts
   576→  const counts = {
   577→    all: clientOrders.length,
   578→    active: clientOrders.filter((o) => o.status !== 'delivered').length,
   579→    delivered: clientOrders.filter((o) => o.status === 'delivered').length,
   580→  };
   581→
   582→  // Stats for header
   583→  const inEscrow = clientOrders.filter((o) => o.escrowStatus === 'held').length;
   584→  const totalEscrowAmount = clientOrders
   585→    .filter((o) => o.escrowStatus === 'held')
   586→    .reduce((sum, o) => sum + o.totalAmount, 0);
   587→
   588→  const filters: { key: FilterKey; label: string }[] = [
   589→    { key: 'all', label: 'Toutes' },
   590→    { key: 'active', label: 'En cours' },
   591→    { key: 'delivered', label: 'Livrées' },
   592→  ];
   593→
   594→  const orderToConfirm = confirmingReceiptId
   595→    ? clientOrders.find((o) => o.id === confirmingReceiptId) || null
   596→    : null;
   597→
   598→  const handleConfirmReceipt = (orderId: string) => {
   599→    setConfirmingReceiptId(orderId);
   600→  };
   601→
   602→  const handleConfirmModalConfirm = () => {
   603→    if (orderToConfirm) {
   604→      confirmReceipt(orderToConfirm.id);
   605→      setShowSuccessToast(true);
   606→      setTimeout(() => setShowSuccessToast(false), 100);
   607→      toast({
   608→        title: 'Réception confirmée',
   609→        description: `L'escrow a été libéré. ${orderToConfirm.vendorName} a reçu les fonds.`,
   610→      });
   611→    }
   612→  };
   613→
   614→  const handleConfirmModalCancel = () => {
   615→    setConfirmingReceiptId(null);
   616→  };
   617→
   618→  const handleToggleExpand = (orderId: string) => {
   619→    setExpandedOrder(expandedOrder === orderId ? null : orderId);
   620→  };
   621→
   622→  return (
   623→    <div className="pb-6">
   624→      {/* Header */}
   625→      <div className="px-4 pt-4 pb-3">
   626→        <h2 className="text-lg font-bold text-gray-900">Mes Commandes</h2>
   627→        <p className="text-xs text-gray-400 mt-0.5">
   628→          Suivez vos achats et confirmez la réception
   629→        </p>
   630→      </div>
   631→
   632→      {/* Escrow Summary Card */}
   633→      {inEscrow > 0 && (
   634→        <div className="px-4 mb-4">
   635→          <div className="bg-gradient-to-br from-amber-50 to-orange-50 rounded-2xl p-4 border border-amber-100">
   636→            <div className="flex items-center gap-3">
   637→              <div className="w-11 h-11 rounded-xl bg-amber-100 flex items-center justify-center flex-shrink-0 relative">
   638→                <Lock className="w-5 h-5 text-amber-600" />
   639→                <div className="absolute -top-0.5 -right-0.5 w-2.5 h-2.5 bg-amber-400 rounded-full animate-pulse" />
   640→              </div>
   641→              <div className="flex-1">
   642→                <span className="text-xs font-bold text-amber-800 block">
   643→                  {inEscrow} commande{inEscrow > 1 ? 's' : ''} en Escrow
   644→                </span>
   645→                <span className="text-[11px] text-amber-600">
   646→                  {formatPrice(totalEscrowAmount)} sécurisés
   647→                </span>
   648→              </div>
   649→              <Shield className="w-5 h-5 text-amber-300" />
   650→            </div>
   651→          </div>
   652→        </div>
   653→      )}
   654→
   655→      {/* Filter Tabs */}
   656→      <div className="px-4 mb-4">
   657→        <div className="flex gap-2 bg-gray-50 p-1 rounded-2xl">
   658→          {filters.map((f) => (
   659→            <button
   660→              key={f.key}
   661→              onClick={() => setClientOrderFilter(f.key)}
   662→              className={`flex-1 py-2 rounded-xl text-xs font-bold transition-all ${
   663→                activeClientOrderFilter === f.key
   664→                  ? 'bg-white text-orange-600 shadow-sm'
   665→                  : 'text-gray-500 hover:text-gray-700'
   666→              }`}
   667→            >
   668→              {f.label}
   669→              <span
   670→                className={`ml-1.5 text-[10px] ${
   671→                  activeClientOrderFilter === f.key
   672→                    ? 'text-orange-400'
   673→                    : 'text-gray-300'
   674→                }`}
   675→              >
   676→                {counts[f.key]}
   677→              </span>
   678→            </button>
   679→          ))}
   680→        </div>
   681→      </div>
   682→
   683→      {/* Orders List */}
   684→      <div className="px-4 space-y-3">
   685→        {filteredOrders.map((order) => (
   686→          <OrderCard
   687→            key={order.id}
   688→            order={order}
   689→            onConfirmReceipt={handleConfirmReceipt}
   690→            onToggleExpand={handleToggleExpand}
   691→            isExpanded={expandedOrder === order.id}
   692→          />
   693→        ))}
   694→
   695→        {filteredOrders.length === 0 && (
   696→          <div className="text-center py-16">
   697→            <div className="w-20 h-20 rounded-2xl bg-gray-50 flex items-center justify-center mx-auto mb-4">
   698→              <ShoppingBag className="w-10 h-10 text-gray-300" />
   699→            </div>
   700→            <h3 className="text-base font-semibold text-gray-700 mb-1">
   701→              {activeClientOrderFilter === 'delivered'
   702→                ? 'Aucune commande livrée'
   703→                : activeClientOrderFilter === 'active'
   704→                ? 'Aucune commande en cours'
   705→                : 'Aucune commande'}
   706→            </h3>
   707→            <p className="text-xs text-gray-400">
   708→              {activeClientOrderFilter === 'all'
   709→                ? 'Vos achats apparaîtront ici'
   710→                : 'Essayez un autre filtre'}
   711→            </p>
   712→          </div>
   713→        )}
   714→      </div>
   715→
   716→      {/* Confirm Receipt Modal */}
   717→      <ConfirmReceiptModal
   718→        order={orderToConfirm}
   719→        onConfirm={handleConfirmModalConfirm}
   720→        onCancel={handleConfirmModalCancel}
   721→      />
   722→    </div>
   723→  );
   724→}
