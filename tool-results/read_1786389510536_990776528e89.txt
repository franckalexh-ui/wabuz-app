    26→const WAVE_COLOR = '#1DC3E0';

    31→// ── Escrow Timeline Step ─────────────────────────────────────
    32→interface TimelineStep {
    33→  label: string;
    34→  detail: string;
    35→  status: 'done' | 'active' | 'pending';
    36→  icon: React.ReactNode;
    37→}
    38→
    39→function EscrowTimeline({ steps }: { steps: TimelineStep[] }) {
    40→  return (
    41→    <div className="w-full space-y-0">
    42→      {steps.map((step, i) => (
    43→        <div key={i} className="flex gap-3">
    44→          {/* Vertical line + dot */}
    45→          <div className="flex flex-col items-center">
    46→            <div
    47→              className={`w-8 h-8 rounded-full flex items-center justify-center flex-shrink-0 transition-all duration-500 ${
    48→                step.status === 'done'
    49→                  ? 'bg-emerald-500'
    50→                  : step.status === 'active'
    51→                  ? 'bg-orange-500 ring-4 ring-orange-100'
    52→                  : 'bg-gray-100'
    53→              }`}
    54→            >
    55→              {step.status === 'active' ? (
    56→                <Loader2 className="w-4 h-4 text-white animate-spin" />
    57→              ) : step.status === 'done' ? (
    58→                <CheckCircle2 className="w-4 h-4 text-white" />
    59→              ) : (
    60→                <Clock className="w-4 h-4 text-gray-300" />
    61→              )}
    62→            </div>
    63→            {i < steps.length - 1 && (
    64→              <div
    65→                className={`w-0.5 h-8 transition-colors duration-500 ${
    66→                  step.status === 'done' ? 'bg-emerald-300' : 'bg-gray-100'
    67→                }`}
    68→              />
    69→            )}
    70→          </div>
    71→          {/* Text */}
    72→          <div className="pb-4">
    73→            <p
    74→              className={`text-sm font-semibold transition-colors ${
    75→                step.status === 'done'
    76→                  ? 'text-emerald-600'
    77→                  : step.status === 'active'
    78→                  ? 'text-gray-900'
    79→                  : 'text-gray-300'
    80→              }`}
    81→            >
    82→              {step.label}
    83→            </p>
    84→            <p
    85→              className={`text-[11px] mt-0.5 transition-colors ${
    86→                step.status === 'pending' ? 'text-gray-200' : 'text-gray-400'
    87→              }`}
    88→            >
    89→              {step.detail}
    90→            </p>
    91→          </div>
    92→        </div>
    93→      ))}
    94→    </div>
    95→  );
    96→}

    99→function EscrowShield({ status }: { status: EscrowStatus }) {
   100→  const isActive = status === 'held' || status === 'collecting';
   101→  return (
   102→    <div className="relative">
   103→      <div
   104→        className={`w-20 h-20 rounded-2xl flex items-center justify-center transition-all duration-700 ${
   105→          status === 'held'
   106→            ? 'bg-amber-100 shadow-lg shadow-amber-200/50'
   107→            : status === 'released'
   108→            ? 'bg-emerald-100 shadow-lg shadow-emerald-200/50'
   109→            : 'bg-gray-100'
   110→        }`}
   111→      >
   112→        {status === 'released' ? (
   113→          <CheckCircle2 className="w-10 h-10 text-emerald-500" />
   114→        ) : (
   115→          <Lock
   116→            className={`w-10 h-10 transition-colors duration-500 ${
   117→              isActive ? 'text-amber-500' : 'text-gray-300'
   118→            }`}
   119→          />
   120→        )}
   121→      </div>
   122→      {isActive && (
   123→        <div className="absolute -top-1 -right-1 w-5 h-5 bg-amber-400 rounded-full flex items-center justify-center">
   124→          <div className="w-2 h-2 bg-white rounded-full animate-pulse" />
   125→        </div>
   126→      )}
   127→    </div>
   128→  );
   129→}

   131→// ── Main Component ───────────────────────────────────────────
   132→export function CheckoutFlow() {
   133→  const {
   134→    cart,
   135→    deliveryZone,
   136→    paymentMethod,
   137→    setPaymentStatus,
   138→    paymentStatus,
   139→    escrowStatus,
   140→    setEscrowStatus,
   141→    setLastOrderId,
   142→    lastOrderId,
   143→    setView,
   144→    clearCart,
   145→    resetCheckout,
   146→    addClientOrder,
   147→    setClientProfile,
   148→    clientPhone: storedPhone,
   149→    clientFirstName: storedFirstName,
   150→    clientLastName: storedLastName,
   151→    clientLoggedIn,
   152→  } = useAppStore();
   153→
   154→  const [step, setStep] = useState<
   155→    'confirming' | 'processing' | 'escrow-held' | 'success'
   156→  >(paymentStatus === 'success' ? 'success' : 'confirming');
   157→  const [showEscrowDetail, setShowEscrowDetail] = useState(false);
   158→  const [processingProgress, setProcessingProgress] = useState(0);
   159→
   160→  // ── OTP / Phone Verification State ──────────────────────
   161→  const [otpCode, setOtpCode] = useState<string | null>(null);
   162→  const [otpInput, setOtpInput] = useState('');
   163→  const [otpSent, setOtpSent] = useState(false);
   164→  const [phoneVerified, setPhoneVerified] = useState(clientLoggedIn && !!storedPhone);
   165→  const [otpError, setOtpError] = useState<string | null>(null);
   166→  const [showOtpBanner, setShowOtpBanner] = useState(false);
   167→
   168→  // ── Client Profile State (Step 3) ───────────────────────
   169→  const [clientFirstName, setClientFirstName] = useState(storedFirstName || '');
   170→  const [clientLastName, setClientLastName] = useState(storedLastName || '');
   171→  const isProfileComplete = clientFirstName.trim().length >= 2;
   172→
   173→  // ── Pre-fill phone from stored profile ───────────────────
   174→  const [phoneNumber, setPhoneNumber] = useState(storedPhone || '');
   175→
   176→  // Validate Ivorian phone format: 07/05/01 XX XX XX XX
   177→  const isPhoneValid = /^(07|05|01)\s?\d{2}\s?\d{2}\s?\d{2}\s?\d{2}$/.test(phoneNumber.replace(/\s/g, ''));
   178→  const isOtpValid = otpInput.length === 4 && /^\d{4}$/.test(otpInput);
   179→
   180→  const handleSendOtp = useCallback(() => {
   181→    if (!isPhoneValid) return;
   182→    const code = String(Math.floor(1000 + Math.random() * 9000));
   183→    setOtpCode(code);
   184→    setOtpSent(true);
   185→    setOtpError(null);
   186→    setOtpInput('');
   187→    setPhoneVerified(false);
   188→    setShowOtpBanner(true);
   189→    // Auto-hide banner after 15 seconds
   190→    setTimeout(() => setShowOtpBanner(false), 15000);
   191→  }, [isPhoneValid]);
   192→
   193→  const handleVerifyOtp = useCallback(() => {
   194→    if (!otpCode || !isOtpValid) return;
   195→    if (otpInput === otpCode) {
   196→      setPhoneVerified(true);
   197→      setOtpError(null);
   198→      setShowOtpBanner(false);
   199→    } else {
   200→      setOtpError('Code incorrect. Réessayez.');
   201→    }
   202→  }, [otpCode, otpInput, isOtpValid]);
   203→
   204→  const itemsTotal = cart.reduce(
   205→    (sum, item) => sum + item.product.price * item.quantity,
   206→    0,
   207→  );
   208→  const total = itemsTotal + (cart.length > 0 ? DELIVERY_FEE : 0);
   209→  const isWave = paymentMethod === 'wave';
   210→  const brandColor = isWave ? WAVE_COLOR : OM_COLOR;
   211→  const brandBg = isWave ? WAVE_BG : OM_BG;
   212→
   213→  // ── Step 2: Processing with progress bar ────────────────
   214→  useEffect(() => {
   215→    if (step !== 'processing') return;
   216→
   217→    setPaymentStatus('processing');
   218→    setEscrowStatus('collecting');
   219→
   220→    const progressInterval = setInterval(() => {
   221→      setProcessingProgress((prev) => {
   222→        if (prev >= 90) {
   223→          clearInterval(progressInterval);
   224→          return 90;
   225→        }
   226→        return prev + Math.random() * 15 + 5;
   227→      });
   228→    }, 400);
   229→
   230→    const timer = setTimeout(() => {
   231→      clearInterval(progressInterval);
   232→      setProcessingProgress(100);
   233→      setEscrowStatus('held');
   234→      const newOrderId = `WAB-${Date.now().toString(36).toUpperCase().slice(-6)}`;
   235→      setLastOrderId(newOrderId);
   236→
   237→      // ── Persist each order to Supabase, then add to local store ──
   238→      // We do the inserts first so we can capture the returned UUID and store
   239→      // it on the local ClientOrder. That UUID is later used by the
   240→      // "Confirmer la réception" button to release the escrow in Supabase.
   241→      (async () => {
   242→        const now = new Date().toISOString();
   243→
   244→        for (let idx = 0; idx < cart.length; idx++) {
   245→          const item = cart[idx];
   246→          const localId = idx === 0 ? newOrderId : `${newOrderId}-${idx + 1}`;
   247→          const totalAmount = item.product.price * item.quantity + DELIVERY_FEE;
   248→
   249→          const { data, error } = await supabase
   250→            .from('orders')
   251→            .insert([
   252→              {
   253→                product_id: item.product.id,
   254→                store_id: item.product.vendorId, // for products fetched from Supabase this is the store UUID
   255→                client_phone: phoneVerified ? `225${phoneNumber.replace(/\s/g, '')}` : '2250700000000',
   256→                delivery_zone: deliveryZone,
   257→                total_amount: totalAmount,
   258→                status: 'paid', // paiement confirmé
   259→                escrow_status: 'held', // argent bloqué en Escrow
   260→                payment_method: paymentMethod, // 'wave' ou 'orange_money'
   261→              },
   262→            ])
   263→            .select();
   264→
   265→          if (error) {
   266→            console.error(
   267→              "Erreur lors de l'enregistrement de la commande:",
   268→              error,
   269→            );
   270→            toast({
   271→              title: 'Commande non sauvegardée',
   272→              description: `Une erreur est survenue pendant l'enregistrement de la commande (${item.product.name}).`,
   273→              variant: 'destructive',
   274→            });
   275→            // Still add to local store so the user sees the order, but without
   276→            // a supabaseId — confirmation will only update local state.
   277→            addClientOrder({
   278→              id: localId,
   279→              productName: item.product.name,
   280→              productImage: item.product.images[0],
   281→              vendorName: item.product.vendorName,
   282→              vendorPhone: item.product.vendorPhone,
   283→              deliveryZone,
   284→              totalAmount,
   285→              status: 'paid',
   286→              escrowStatus: 'held',
   287→              paymentMethod,
   288→              quantity: item.quantity,
   289→              createdAt: now,
   290→            });
   291→          } else {
   292→            console.log(
   293→              'Commande enregistrée avec succès dans Supabase pour',
   294→              item.product.name,
   295→            );
   296→            const insertedRow = Array.isArray(data) && data.length > 0 ? data[0] : null;
   297→            addClientOrder({
   298→              id: localId,
   299→              productName: item.product.name,
   300→              productImage: item.product.images[0],
   301→              vendorName: item.product.vendorName,
   302→              vendorPhone: item.product.vendorPhone,
   303→              deliveryZone,
   304→              totalAmount,
   305→              status: 'paid',
   306→              escrowStatus: 'held',
   307→              paymentMethod,
   308→              quantity: item.quantity,
   309→              createdAt: now,
   310→              supabaseId: insertedRow?.id ? String(insertedRow.id) : undefined,
   311→            });
   312→          }
   313→        }
   314→      })();
   315→
   316→      setTimeout(() => setStep('escrow-held'), 600);
   317→    }, 3500);
   318→
   319→    return () => {
   320→      clearTimeout(timer);
   321→      clearInterval(progressInterval);
   322→    };
   323→  }, [step, setPaymentStatus, setEscrowStatus, setLastOrderId, cart, deliveryZone, paymentMethod, addClientOrder]);
   324→
   325→  const handleConfirmPayment = useCallback(() => {
   326→    // Save client profile to store + localStorage before processing
   327→    if (phoneVerified && isProfileComplete && phoneNumber) {
   328→      setClientProfile(
   329→        `225${phoneNumber.replace(/\s/g, '')}`,
   330→        clientFirstName.trim(),
   331→        clientLastName.trim()
   332→      );
   333→    }
   334→    setProcessingProgress(0);
   335→    setStep('processing');
   336→  }, [phoneVerified, isProfileComplete, phoneNumber, clientFirstName, clientLastName, setClientProfile]);
   337→
   338→  const handleContinueShopping = useCallback(() => {
   339→    clearCart();
   340→    setView('home');
   341→    resetCheckout();
   342→  }, [clearCart, setView, resetCheckout]);
   343→
   344→  // ── Escrow Timeline Steps based on current status ────────
   345→  const getEscrowSteps = (): TimelineStep[] => [
   346→    {
   347→      label: 'Paiement initié',
   348→      detail: isWave
   349→        ? 'Envoi de la demande Wave...'
   350→        : 'Envoi de la demande Orange Money...',
   351→      status: 'done',
   352→      icon: <CheckCircle2 className="w-4 h-4" />,
   353→    },
   354→    {
   355→      label: 'Argent collecté',
   356→      detail: `${formatPrice(total)} prélevé sur votre compte`,
   357→      status:
   358→        escrowStatus === 'collecting'
   359→          ? 'active'
   360→          : ['held', 'released'].includes(escrowStatus)
   361→          ? 'done'
   362→          : 'pending',
   363→      icon: <CheckCircle2 className="w-4 h-4" />,
   364→    },
   365→    {
   366→      label: 'Escrow WABUZ actif',
   367→      detail: 'Fonds bloqués en sécurité jusqu\'à la livraison',
   368→      status:
   369→        escrowStatus === 'held'
   370→          ? 'active'
   371→          : escrowStatus === 'released'
   372→          ? 'done'
   373→          : 'pending',
   374→      icon: <Lock className="w-4 h-4" />,
   375→    },
   376→    {
   377→      label: 'Livraison confirmée',
   378→      detail: 'Le vendeur reçoit l\'argent après votre confirmation',
   379→      status: escrowStatus === 'released' ? 'done' : 'pending',
   380→      icon: <Package className="w-4 h-4" />,
   381→    },
   382→  ];
   383→
   384→  // ══════════════════════════════════════════════════════════
   385→  // SUCCESS VIEW

   387→  if (step === 'success') {
   388→    return (
   389→      <div className="min-h-[75vh] flex flex-col items-center justify-center px-6 text-center">
   390→        {/* Success Animation */}
   391→        <div className="relative mb-6">
   392→          <div className="w-24 h-24 rounded-full bg-emerald-100 flex items-center justify-center animate-in zoom-in duration-500">
   393→            <CheckCircle2 className="w-14 h-14 text-emerald-500" />
   394→          </div>
   395→          <div className="absolute -top-2 -right-2 w-8 h-8 bg-amber-400 rounded-full flex items-center justify-center animate-in zoom-in duration-700">
   396→            <span className="text-lg">🎉</span>
   397→          </div>
   398→        </div>
   399→
   400→        <h2 className="text-2xl font-extrabold text-gray-900 mb-2">
   401→          {clientFirstName
   402→            ? `Merci ${clientFirstName} !`
   403→            : 'Paiement Confirmé !'}
   404→        </h2>
   405→        <p className="text-sm text-gray-500 mb-2 max-w-xs">
   406→          Votre commande a été enregistrée. Le vendeur sera notifié et vous
   407→          recevrez une confirmation par WhatsApp.
   408→        </p>
   409→
   410→        {/* Order ID Badge */}
   411→        {lastOrderId && (
   412→          <div className="inline-flex items-center gap-2 px-3 py-1.5 rounded-full bg-gray-100 mb-6">
   413→            <Package className="w-3.5 h-3.5 text-gray-400" />
   414→            <span className="text-xs font-mono font-bold text-gray-600">
   415→              {lastOrderId}
   416→            </span>
   417→          </div>
   418→        )}
   419→
   420→        {/* Order Details Card */}
   421→        <div className="w-full bg-gray-50 rounded-2xl p-5 mb-4 text-left">
   422→          <div className="flex items-center gap-2 mb-4">
   423→            <Package className="w-4 h-4 text-orange-500" />
   424→            <span className="text-sm font-semibold text-gray-900">
   425→              Détails de la commande
   426→            </span>
   427→          </div>
   428→          {cart.map((item) => (
   429→            <div
   430→              key={item.product.id}
   431→              className="flex items-center gap-3 mb-3 last:mb-0"
   432→            >
   433→              <div className="w-12 h-12 rounded-lg overflow-hidden bg-gray-200 flex-shrink-0">

   459→              <span className="text-gray-900">Total</span>
   460→              <span className="text-orange-600">{formatPrice(total)}</span>

   462→          </div>

   465→        {/* Escrow Status Card */}

   467→          <div className="flex items-center gap-3">
   468→            <EscrowShield status="held" />
