    41→    41→    <div className="w-full space-y-0">
    42→    42→      {steps.map((step, i) => (
    43→    43→        <div key={i} className="flex gap-3">
    44→    44→          {/* Vertical line + dot */}

    66→    66→                  step.status === 'done' ? 'bg-emerald-300' : 'bg-gray-100'
    67→    67→                }`}
    68→    68→              />
    69→    69→            )}
    70→    70→          </div>
    71→    71→          {/* Text */}
    72→    72→          <div className="pb-4">
    73→    73→            <p
    74→    74→              className={`text-sm font-semibold transition-colors ${
    75→    75→                step.status === 'done'
    76→    76→                  ? 'text-emerald-600'
    77→    77→                  : step.status === 'active'

   131→   131→// ── Main Component ───────────────────────────────────────────
   132→   132→export function CheckoutFlow() {
   133→   133→  const {
   134→   134→    cart,
   135→   135→    deliveryZone,
   136→   136→    paymentMethod,
   137→   137→    setPaymentStatus,
   138→   138→    paymentStatus,
   139→   139→    escrowStatus,
   140→   140→    setEscrowStatus,
   141→   141→    setLastOrderId,
   142→   142→    lastOrderId,
   143→   143→    setView,
   144→   144→    clearCart,
   145→   145→    resetCheckout,
   146→   146→    addClientOrder,
   147→   147→    setClientProfile,
   148→   148→    clientPhone: storedPhone,
   149→   149→    clientFirstName: storedFirstName,
   150→   150→    clientLastName: storedLastName,
   151→   151→    clientLoggedIn,
   152→   152→  } = useAppStore();
   153→   153→
   154→   154→  const [step, setStep] = useState<
   155→   155→    'confirming' | 'processing' | 'escrow-held' | 'success'
   156→   156→  >(paymentStatus === 'success' ? 'success' : 'confirming');
   157→   157→  const [showEscrowDetail, setShowEscrowDetail] = useState(false);
   158→   158→  const [processingProgress, setProcessingProgress] = useState(0);
   159→   159→
   160→   160→  // ── OTP / Phone Verification State ──────────────────────
   161→   161→  const [otpCode, setOtpCode] = useState<string | null>(null);
   162→   162→  const [otpInput, setOtpInput] = useState('');
   163→   163→  const [otpSent, setOtpSent] = useState(false);
   164→   164→  const [phoneVerified, setPhoneVerified] = useState(clientLoggedIn && !!storedPhone);
   165→   165→  const [otpError, setOtpError] = useState<string | null>(null);
   166→   166→  const [showOtpBanner, setShowOtpBanner] = useState(false);
   167→   167→
   168→   168→  // ── Client Profile State (Step 3) ───────────────────────
   169→   169→  const [clientFirstName, setClientFirstName] = useState(storedFirstName || '');
   170→   170→  const [clientLastName, setClientLastName] = useState(storedLastName || '');
   171→   171→  const isProfileComplete = clientFirstName.trim().length >= 2;
   172→   172→
   173→   173→  // ── Pre-fill phone from stored profile ───────────────────
   174→   174→  const [phoneNumber, setPhoneNumber] = useState(storedPhone || '');
   175→   175→
   176→   176→  // Validate Ivorian phone format: 07/05/01 XX XX XX XX
   177→   177→  const isPhoneValid = /^(07|05|01)\s?\d{2}\s?\d{2}\s?\d{2}\s?\d{2}$/.test(phoneNumber.replace(/\s/g, ''));
   178→   178→  const isOtpValid = otpInput.length === 4 && /^\d{4}$/.test(otpInput);
   179→   179→
   180→   180→  const handleSendOtp = useCallback(() => {
   181→   181→    if (!isPhoneValid) return;
   182→   182→    const code = String(Math.floor(1000 + Math.random() * 9000));
   183→   183→    setOtpCode(code);
   184→   184→    setOtpSent(true);
   185→   185→    setOtpError(null);
   186→   186→    setOtpInput('');
   187→   187→    setPhoneVerified(false);
   188→   188→    setShowOtpBanner(true);
   189→   189→    // Auto-hide banner after 15 seconds
   190→   190→    setTimeout(() => setShowOtpBanner(false), 15000);
   191→   191→  }, [isPhoneValid]);
   192→   192→
   193→   193→  const handleVerifyOtp = useCallback(() => {
   194→   194→    if (!otpCode || !isOtpValid) return;
   195→   195→    if (otpInput === otpCode) {
   196→   196→      setPhoneVerified(true);
   197→   197→      setOtpError(null);
   198→   198→      setShowOtpBanner(false);
   199→   199→    } else {
   200→   200→      setOtpError('Code incorrect. Réessayez.');
   201→   201→    }
   202→   202→  }, [otpCode, otpInput, isOtpValid]);
   203→   203→
   204→   204→  const itemsTotal = cart.reduce(
   205→   205→    (sum, item) => sum + item.product.price * item.quantity,
   206→   206→    0,
   207→   207→  );
   208→   208→  const total = itemsTotal + (cart.length > 0 ? DELIVERY_FEE : 0);
   209→   209→  const isWave = paymentMethod === 'wave';
   210→   210→  const brandColor = isWave ? WAVE_COLOR : OM_COLOR;
   211→   211→  const brandBg = isWave ? WAVE_BG : OM_BG;
   212→   212→
   213→   213→  // ── Step 2: Processing with progress bar ────────────────
   214→   214→  useEffect(() => {
   215→   215→    if (step !== 'processing') return;
   216→   216→
   217→   217→    setPaymentStatus('processing');
   218→   218→    setEscrowStatus('collecting');
   219→   219→
   220→   220→    const progressInterval = setInterval(() => {
   221→   221→      setProcessingProgress((prev) => {
   222→   222→        if (prev >= 90) {
   223→   223→          clearInterval(progressInterval);
   224→   224→          return 90;
   225→   225→        }
   226→   226→        return prev + Math.random() * 15 + 5;
   227→   227→      });
   228→   228→    }, 400);
   229→   229→
   230→   230→    const timer = setTimeout(() => {
   231→   231→      clearInterval(progressInterval);
   232→   232→      setProcessingProgress(100);
   233→   233→      setEscrowStatus('held');
   234→   234→      const newOrderId = `WAB-${Date.now().toString(36).toUpperCase().slice(-6)}`;
   235→   235→      setLastOrderId(newOrderId);
   236→   236→
   237→   237→      // ── Persist each order to Supabase, then add to local store ──
   238→   238→      // We do the inserts first so we can capture the returned UUID and store
   239→   239→      // it on the local ClientOrder. That UUID is later used by the
   240→   240→      // "Confirmer la réception" button to release the escrow in Supabase.
   241→   241→      (async () => {
   242→   242→        const now = new Date().toISOString();
   243→   243→
   244→   244→        for (let idx = 0; idx < cart.length; idx++) {
   245→   245→          const item = cart[idx];
   246→   246→          const localId = idx === 0 ? newOrderId : `${newOrderId}-${idx + 1}`;
   247→   247→          const totalAmount = item.product.price * item.quantity + DELIVERY_FEE;
   248→   248→
   249→   249→          const { data, error } = await supabase
   250→   250→            .from('orders')
   251→   251→            .insert([
   252→   252→              {
   253→   253→                product_id: item.product.id,
   254→   254→                store_id: item.product.vendorId, // for products fetched from Supabase this is the store UUID
   255→   255→                client_phone: phoneVerified ? `225${phoneNumber.replace(/\s/g, '')}` : '2250700000000',
   256→   256→                delivery_zone: deliveryZone,
   257→   257→                total_amount: totalAmount,
   258→   258→                status: 'paid', // paiement confirmé
   259→   259→                escrow_status: 'held', // argent bloqué en Escrow
   260→   260→                payment_method: paymentMethod, // 'wave' ou 'orange_money'
   261→   261→              },
   262→   262→            ])
   263→   263→            .select();
   264→   264→
   265→   265→          if (error) {
   266→   266→            console.error(
   267→   267→              "Erreur lors de l'enregistrement de la commande:",
   268→   268→              error,
   269→   269→            );
   270→   270→            toast({
   271→   271→              title: 'Commande non sauvegardée',
   272→   272→              description: `Une erreur est survenue pendant l'enregistrement de la commande (${item.product.name}).`,
   273→   273→              variant: 'destructive',
   274→   274→            });
   275→   275→            // Still add to local store so the user sees the order, but without
   276→   276→            // a supabaseId — confirmation will only update local state.
   277→   277→            addClientOrder({
   278→   278→              id: localId,
   279→   279→              productName: item.product.name,
   280→   280→              productImage: item.product.images[0],
   281→   281→              vendorName: item.product.vendorName,
   282→   282→              vendorPhone: item.product.vendorPhone,
   283→   283→              deliveryZone,
   284→   284→              totalAmount,
   285→   285→              status: 'paid',
   286→   286→              escrowStatus: 'held',
   287→   287→              paymentMethod,
   288→   288→              quantity: item.quantity,
   289→   289→              createdAt: now,
   290→   290→            });
   291→   291→          } else {
   292→   292→            console.log(
   293→   293→              'Commande enregistrée avec succès dans Supabase pour',
   294→   294→              item.product.name,
   295→   295→            );
   296→   296→            const insertedRow = Array.isArray(data) && data.length > 0 ? data[0] : null;
   297→   297→            addClientOrder({
   298→   298→              id: localId,
   299→   299→              productName: item.product.name,
   300→   300→              productImage: item.product.images[0],
   301→   301→              vendorName: item.product.vendorName,
   302→   302→              vendorPhone: item.product.vendorPhone,
   303→   303→              deliveryZone,
   304→   304→              totalAmount,
   305→   305→              status: 'paid',
   306→   306→              escrowStatus: 'held',
   307→   307→              paymentMethod,
   308→   308→              quantity: item.quantity,
   309→   309→              createdAt: now,
   310→   310→              supabaseId: insertedRow?.id ? String(insertedRow.id) : undefined,
   311→   311→            });
   312→   312→          }
   313→   313→        }
   314→   314→      })();
   315→   315→
   316→   316→      setTimeout(() => setStep('escrow-held'), 600);
   317→   317→    }, 3500);
   318→   318→
   319→   319→    return () => {
   320→   320→      clearTimeout(timer);
   321→   321→      clearInterval(progressInterval);
   322→   322→    };
   323→   323→  }, [step, setPaymentStatus, setEscrowStatus, setLastOrderId, cart, deliveryZone, paymentMethod, addClientOrder]);
   324→   324→
   325→   325→  const handleConfirmPayment = useCallback(() => {
   326→   326→    // Save client profile to store + localStorage before processing
   327→   327→    if (phoneVerified && isProfileComplete && phoneNumber) {
   328→   328→      setClientProfile(
   329→   329→        `225${phoneNumber.replace(/\s/g, '')}`,
   330→   330→        clientFirstName.trim(),
   331→   331→        clientLastName.trim()
   332→   332→      );
   333→   333→    }
   334→   334→    setProcessingProgress(0);
   335→   335→    setStep('processing');
   336→   336→  }, [phoneVerified, isProfileComplete, phoneNumber, clientFirstName, clientLastName, setClientProfile]);
   337→   337→
   338→   338→  const handleContinueShopping = useCallback(() => {
   339→   339→    clearCart();
   340→   340→    setView('home');
   341→   341→    resetCheckout();
   342→   342→  }, [clearCart, setView, resetCheckout]);
   343→   343→
   344→   344→  // ── Escrow Timeline Steps based on current status ────────
   345→   345→  const getEscrowSteps = (): TimelineStep[] => [
   346→   346→    {
   347→   347→      label: 'Paiement initié',
   348→   348→      detail: isWave
   349→   349→        ? 'Envoi de la demande Wave...'
   350→   350→        : 'Envoi de la demande Orange Money...',
   351→   351→      status: 'done',
   352→   352→      icon: <CheckCircle2 className="w-4 h-4" />,
   353→   353→    },
   354→   354→    {
   355→   355→      label: 'Argent collecté',
   356→   356→      detail: `${formatPrice(total)} prélevé sur votre compte`,
   357→   357→      status:
   358→   358→        escrowStatus === 'collecting'
   359→   359→          ? 'active'
   360→   360→          : ['held', 'released'].includes(escrowStatus)
   361→   361→          ? 'done'
   362→   362→          : 'pending',
   363→   363→      icon: <CheckCircle2 className="w-4 h-4" />,
   364→   364→    },
   365→   365→    {
   366→   366→      label: 'Escrow WABUZ actif',
   367→   367→      detail: 'Fonds bloqués en sécurité jusqu\'à la livraison',
   368→   368→      status:
   369→   369→        escrowStatus === 'held'
   370→   370→          ? 'active'
   371→   371→          : escrowStatus === 'released'
   372→   372→          ? 'done'
   373→   373→          : 'pending',
   374→   374→      icon: <Lock className="w-4 h-4" />,
   375→   375→    },
   376→   376→    {
   377→   377→      label: 'Livraison confirmée',
   378→   378→      detail: 'Le vendeur reçoit l\'argent après votre confirmation',
   379→   379→      status: escrowStatus === 'released' ? 'done' : 'pending',
   380→   380→      icon: <Package className="w-4 h-4" />,
   381→   381→    },
   382→   382→  ];
   383→   383→
   384→   384→  // ══════════════════════════════════════════════════════════
   385→   385→  // SUCCESS VIEW
   386→   386→  // ══════════════════════════════════════════════════════════
   387→   387→  if (step === 'success') {
   388→   388→    return (
   389→   389→      <div className="min-h-[75vh] flex flex-col items-center justify-center px-6 text-center">
   390→   390→        {/* Success Animation */}
   391→   391→        <div className="relative mb-6">
   392→   392→          <div className="w-24 h-24 rounded-full bg-emerald-100 flex items-center justify-center animate-in zoom-in duration-500">
   393→   393→            <CheckCircle2 className="w-14 h-14 text-emerald-500" />
   394→   394→          </div>
   395→   395→          <div className="absolute -top-2 -right-2 w-8 h-8 bg-amber-400 rounded-full flex items-center justify-center animate-in zoom-in duration-700">
   396→   396→            <span className="text-lg">🎉</span>
   397→   397→          </div>
   398→   398→        </div>
   399→   399→
   400→   400→        <h2 className="text-2xl font-extrabold text-gray-900 mb-2">
   401→   401→          {clientFirstName
   402→   402→            ? `Merci ${clientFirstName} !`
   403→   403→            : 'Paiement Confirmé !'}
   404→   404→        </h2>
   405→   405→        <p className="text-sm text-gray-500 mb-2 max-w-xs">
   406→   406→          Votre commande a été enregistrée. Le vendeur sera notifié et vous
   407→   407→          recevrez une confirmation par WhatsApp.
   408→   408→        </p>
   409→   409→
   410→   410→        {/* Order ID Badge */}
   411→   411→        {lastOrderId && (
   412→   412→          <div className="inline-flex items-center gap-2 px-3 py-1.5 rounded-full bg-gray-100 mb-6">
   413→   413→            <Package className="w-3.5 h-3.5 text-gray-400" />
   414→   414→            <span className="text-xs font-mono font-bold text-gray-600">
   415→   415→              {lastOrderId}
   416→   416→            </span>
   417→   417→          </div>
   418→   418→        )}
   419→   419→
   420→   420→        {/* Order Details Card */}
   421→   421→        <div className="w-full bg-gray-50 rounded-2xl p-5 mb-4 text-left">
   422→   422→          <div className="flex items-center gap-2 mb-4">
   423→   423→            <Package className="w-4 h-4 text-orange-500" />
   424→   424→            <span className="text-sm font-semibold text-gray-900">
   425→   425→              Détails de la commande
   426→   426→            </span>
   427→   427→          </div>
   428→   428→          {cart.map((item) => (
   429→   429→            <div
   430→   430→              key={item.product.id}
   431→   431→              className="flex items-center gap-3 mb-3 last:mb-0"
   432→   432→            >
   433→   433→              <div className="w-12 h-12 rounded-lg overflow-hidden bg-gray-200 flex-shrink-0">
   434→   434→                <img
   435→   435→                  src={item.product.images[0]}
   436→   436→                  alt={item.product.name}
   437→   437→                  className="w-full h-full object-cover"
   438→   438→                />
   439→   439→              </div>
   440→   440→              <div className="flex-1 min-w-0">
   441→   441→                <p className="text-sm font-medium text-gray-900 truncate">
   442→   442→                  {item.product.name}
   443→   443→                </p>
   444→   444→                <p className="text-xs text-gray-400">x{item.quantity}</p>
   445→   445→              </div>
   446→   446→              <span className="text-sm font-bold text-gray-700">
   447→   447→                {formatPrice(item.product.price * item.quantity)}
   448→   448→              </span>
   449→   449→            </div>
   450→   450→          ))}
   451→   451→          <div className="border-t border-gray-200 mt-3 pt-3 space-y-1.5">
   452→   452→            <div className="flex justify-between text-xs">
   453→   453→              <span className="text-gray-500">Livraison à {deliveryZone}</span>
