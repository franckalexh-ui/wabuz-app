    79→    79→function OrderStepper({ currentStatus }: { currentStatus: ClientOrderStatus }) {
    80→    80→  const currentStep = STATUS_CONFIG[currentStatus].step;
    81→    81→  const steps = [
    82→    82→    { num: 1, label: 'Commandé', icon: <ShoppingBag className="w-3 h-3" /> },
    83→    83→    { num: 2, label: 'Payé', icon: <CreditCard className="w-3 h-3" /> },
    84→    84→    { num: 3, label: 'Expédié', icon: <Truck className="w-3 h-3" /> },
    85→    85→    { num: 4, label: 'Livré', icon: <Package className="w-3 h-3" /> },
    86→    86→  ];
    87→    87→
    88→    88→  return (
    89→    89→    <div className="flex items-center justify-between">
    90→    90→      {steps.map((step, i) => {
    91→    91→        const isDone = step.num <= currentStep;
    92→    92→        const isCurrent = step.num === currentStep;
    93→    93→        return (
    94→    94→          <div key={step.num} className="flex items-center flex-1 last:flex-none">
    95→    95→            <div className="flex flex-col items-center gap-1">
    96→    96→              <div
    97→    97→                className={`w-7 h-7 rounded-full flex items-center justify-center transition-all duration-500 ${
    98→    98→                  isDone
    99→    99→                    ? isCurrent
   100→   100→                      ? 'bg-orange-500 text-white ring-4 ring-orange-100'
   101→   101→                      : 'bg-emerald-500 text-white'
   102→   102→                    : 'bg-gray-100 text-gray-300'
   103→   103→                }`}
   104→   104→              >
   105→   105→                {isDone && !isCurrent ? (
   106→   106→                  <CheckCircle2 className="w-3.5 h-3.5" />
   107→   107→                ) : (
   108→   108→                  step.icon
   109→   109→                )}
   110→   110→              </div>
   111→   111→              <span
   112→   112→                className={`text-[9px] font-medium transition-colors ${
   113→   113→                  isDone ? 'text-gray-700' : 'text-gray-300'
   114→   114→                }`}
   115→   115→              >
   116→   116→                {step.label}
   117→   117→              </span>
   118→   118→            </div>

   120→   120→              <div
   121→   121→                className={`flex-1 h-0.5 mx-1 mb-4 transition-colors duration-500 ${
   122→   122→                  step.num < currentStep ? 'bg-emerald-300' : 'bg-gray-100'

   124→   124→              />
   125→   125→            )}
   126→   126→          </div>
   127→   127→        );
   128→   128→      })}

   552→   552→  const {

   554→   554→    activeClientOrderFilter,
   555→   555→    setClientOrderFilter,

   566→   566→    if (activeClientOrderFilter === 'active') {
