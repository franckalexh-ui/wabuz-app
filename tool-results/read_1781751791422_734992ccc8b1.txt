    27→interface TimelineStep {
    28→  label: string;
    29→  detail: string;
    30→  status: 'done' | 'active' | 'pending';
    31→  icon: React.ReactNode;

    33→
    34→function EscrowTimeline({ steps }: { steps: TimelineStep[] }) {
    35→  return (
    36→    <div className="w-full space-y-0">
    37→      {steps.map((step, i) => (
    38→        <div key={i} className="flex gap-3">
    39→          {/* Vertical line + dot */}
    40→          <div className="flex flex-col items-center">
    41→            <div
    42→              className={`w-8 h-8 rounded-full flex items-center justify-center flex-shrink-0 transition-all duration-500 ${
    43→                step.status === 'done'
    44→                  ? 'bg-emerald-500'
    45→                  : step.status === 'active'
    46→                  ? 'bg-orange-500 ring-4 ring-orange-100'
    47→                  : 'bg-gray-100'
    48→              }`}
    49→            >
    50→              {step.status === 'active' ? (
    51→                <Loader2 className="w-4 h-4 text-white animate-spin" />
    52→              ) : step.status === 'done' ? (
    53→                <CheckCircle2 className="w-4 h-4 text-white" />
    54→              ) : (
    55→                <Clock className="w-4 h-4 text-gray-300" />
    56→              )}
    57→            </div>
    58→            {i < steps.length - 1 && (
    59→              <div
    60→                className={`w-0.5 h-8 transition-colors duration-500 ${
    61→                  step.status === 'done' ? 'bg-emerald-300' : 'bg-gray-100'
    62→                }`}
    63→              />
    64→            )}
    65→          </div>
    66→          {/* Text */}
    67→          <div className="pb-4">
    68→            <p
    69→              className={`text-sm font-semibold transition-colors ${
    70→                step.status === 'done'
    71→                  ? 'text-emerald-600'
    72→                  : step.status === 'active'
    73→                  ? 'text-gray-900'
    74→                  : 'text-gray-300'

    76→            >
    77→              {step.label}
    78→            </p>
    79→            <p
    80→              className={`text-[11px] mt-0.5 transition-colors ${
    81→                step.status === 'pending' ? 'text-gray-200' : 'text-gray-400'

    85→            </p>
    86→          </div>
    87→        </div>

    94→function EscrowShield({ status }: { status: EscrowStatus }) {

    96→  return (
    97→    <div className="relative">

    99→        className={`w-20 h-20 rounded-2xl flex items-center justify-center transition-all duration-700 ${

   120→        </div>
