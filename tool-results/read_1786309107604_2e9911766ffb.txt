    28→// ── Escrow Timeline Step ─────────────────────────────────────
    29→interface TimelineStep {
    30→  label: string;
    31→  detail: string;
    32→  status: 'done' | 'active' | 'pending';
    33→  icon: React.ReactNode;
    34→}
    35→
    36→function EscrowTimeline({ steps }: { steps: TimelineStep[] }) {
    37→  return (
    38→    <div className="w-full space-y-0">
    39→      {steps.map((step, i) => (
    40→        <div key={i} className="flex gap-3">
    41→          {/* Vertical line + dot */}
    42→          <div className="flex flex-col items-center">
    43→            <div
    44→              className={`w-8 h-8 rounded-full flex items-center justify-center flex-shrink-0 transition-all duration-500 ${
    45→                step.status === 'done'
    46→                  ? 'bg-emerald-500'
    47→                  : step.status === 'active'
    48→                  ? 'bg-orange-500 ring-4 ring-orange-100'
    49→                  : 'bg-gray-100'
    50→              }`}
    51→            >
    52→              {step.status === 'active' ? (
    53→                <Loader2 className="w-4 h-4 text-white animate-spin" />
    54→              ) : step.status === 'done' ? (
    55→                <CheckCircle2 className="w-4 h-4 text-white" />
    56→              ) : (
    57→                <Clock className="w-4 h-4 text-gray-300" />
    58→              )}
    59→            </div>
    60→            {i < steps.length - 1 && (
    61→              <div
    62→                className={`w-0.5 h-8 transition-colors duration-500 ${
    63→                  step.status === 'done' ? 'bg-emerald-300' : 'bg-gray-100'
    64→                }`}
    65→              />
    66→            )}
    67→          </div>
    68→          {/* Text */}
    69→          <div className="pb-4">
    70→            <p
    71→              className={`text-sm font-semibold transition-colors ${
    72→                step.status === 'done'
    73→                  ? 'text-emerald-600'
    74→                  : step.status === 'active'
    75→                  ? 'text-gray-900'
    76→                  : 'text-gray-300'
    77→              }`}
    78→            >
    79→              {step.label}
    80→            </p>
    81→            <p
    82→              className={`text-[11px] mt-0.5 transition-colors ${
    83→                step.status === 'pending' ? 'text-gray-200' : 'text-gray-400'
    84→              }`}
    85→            >
    86→              {step.detail}
    87→            </p>
    88→          </div>
    89→        </div>
    90→      ))}
    91→    </div>
    92→  );
    93→}

    95→// ── Escrow Shield Animation ──────────────────────────────────
    96→function EscrowShield({ status }: { status: EscrowStatus }) {
    97→  const isActive = status === 'held' || status === 'collecting';
    98→  return (
    99→    <div className="relative">
   100→      <div
   101→        className={`w-20 h-20 rounded-2xl flex items-center justify-center transition-all duration-700 ${
   102→          status === 'held'
   103→            ? 'bg-amber-100 shadow-lg shadow-amber-200/50'
   104→            : status === 'released'
   105→            ? 'bg-emerald-100 shadow-lg shadow-emerald-200/50'
   106→            : 'bg-gray-100'
   107→        }`}
   108→      >
   109→        {status === 'released' ? (
   110→          <CheckCircle2 className="w-10 h-10 text-emerald-500" />
   111→        ) : (
   112→          <Lock
   113→            className={`w-10 h-10 transition-colors duration-500 ${
   114→              isActive ? 'text-amber-500' : 'text-gray-300'
   115→            }`}
   116→          />
   117→        )}
   118→      </div>
   119→      {isActive && (
   120→        <div className="absolute -top-1 -right-1 w-5 h-5 bg-amber-400 rounded-full flex items-center justify-center">
   121→          <div className="w-2 h-2 bg-white rounded-full animate-pulse" />
   122→        </div>
   123→      )}
   124→    </div>
   125→  );
   126→}

   156→  const total = itemsTotal + (cart.length > 0 ? DELIVERY_FEE : 0);
   157→  const isWave = paymentMethod === 'wave';
   158→  const brandColor = isWave ? WAVE_COLOR : OM_COLOR;
   159→  const brandBg = isWave ? WAVE_BG : OM_BG;
   160→

   169→      setProcessingProgress((prev) => {

   172→          return 90;

   177→
   178→    const timer = setTimeout(() => {

   182→      const newOrderId = `WAB-${Date.now().toString(36).toUpperCase().slice(-6)}`;

   184→
   185→      // ── Persist each order to Supabase, then add to local store ──

   194→          const localId = idx === 0 ? newOrderId : `${newOrderId}-${idx + 1}`;
   195→          const totalAmount = item.product.price * item.quantity + DELIVERY_FEE;
   196→
   197→          const { data, error } = await supabase
   198→            .from('orders')
   199→            .insert([
   200→              {
   201→                product_id: item.product.id,
   202→                store_id: item.product.vendorId, // for products fetched from Supabase this is the store UUID
   203→                client_phone: '2250700000000', // numéro factice pour l'instant
   204→                delivery_zone: deliveryZone,
   205→                total_amount: totalAmount,
   206→                status: 'paid', // paiement confirmé
   207→                escrow_status: 'held', // argent bloqué en Escrow
   208→                payment_method: paymentMethod, // 'wave' ou 'orange_money'
   209→              },
   210→            ])
   211→            .select();
   212→
   213→          if (error) {
   214→            console.error(
   215→              "Erreur lors de l'enregistrement de la commande:",
   216→              error,
   217→            );

   219→              title: 'Commande non sauvegardée',
   220→              description: `Une erreur est survenue pendant l'enregistrement de la commande (${item.product.name}).`,
   221→              variant: 'destructive',
   222→            });

   224→            // a supabaseId — confirmation will only update local state.
   225→            addClientOrder({
   226→              id: localId,
   227→              productName: item.product.name,
   228→              productImage: item.product.images[0],
   229→              vendorName: item.product.vendorName,
   230→              vendorPhone: item.product.vendorPhone,
   231→              deliveryZone,
   232→              totalAmount,
   233→              status: 'paid',
   234→              escrowStatus: 'held',
   235→              paymentMethod,
   236→              quantity: item.quantity,
   237→              createdAt: now,
   238→            });
   239→          } else {
   240→            console.log(
   241→              'Commande enregistrée avec succès dans Supabase pour',
   242→              item.product.name,
   243→            );
   244→            const insertedRow = Array.isArray(data) && data.length > 0 ? data[0] : null;
   245→            addClientOrder({
   246→              id: localId,
   247→              productName: item.product.name,
   248→              productImage: item.product.images[0],
   249→              vendorName: item.product.vendorName,
   250→              vendorPhone: item.product.vendorPhone,
   251→              deliveryZone,
   252→              totalAmount,
   253→              status: 'paid',
   254→              escrowStatus: 'held',
   255→              paymentMethod,
   256→              quantity: item.quantity,
   257→              createdAt: now,
   258→              supabaseId: insertedRow?.id ? String(insertedRow.id) : undefined,
   259→            });
   260→          }
